<script>
  const audio = document.getElementById('audio');
  const reproductor = document.getElementById('reproductor');
  const btnPlay = document.getElementById('btnPlay');
  const nombrePista = document.getElementById('nombrePista');
  const estadoPista = document.getElementById('estadoPista');
  const botones = document.querySelectorAll('.boton');

  const pistas = {
    apertura: { archivo: 'musica/apertura.mp3', nombre: 'Apertura de Tenida' },
    marchas:  { archivo: 'musica/marchas.mp3',  nombre: 'Marchas' },
    trabajos: { archivo: 'musica/trabajos.mp3', nombre: 'Trabajos' },
    clausura: { archivo: 'musica/clausura.mp3', nombre: 'Clausura' },
    agape:    { archivo: 'musica/agape.mp3',    nombre: 'Ágape' }
  };

  botones.forEach(btn => {
    btn.addEventListener('click', () => {
      const momento = btn.dataset.momento;
      const pista = pistas[momento];

      botones.forEach(b => b.classList.remove('activo'));
      btn.classList.add('activo');

      audio.src = pista.archivo;
      nombrePista.textContent = pista.nombre;
      reproductor.classList.add('visible');

      audio.play().then(() => {
        estadoPista.textContent = 'Reproduciendo';
        btnPlay.textContent = '❚❚';
      }).catch(err => {
        estadoPista.textContent = 'Error: falta el MP3';
        console.error(err);
      });
    });
  });

  btnPlay.addEventListener('click', () => {
    if (audio.paused) {
      audio.play();
      btnPlay.textContent = '❚❚';
      estadoPista.textContent = 'Reproduciendo';
    } else {
      audio.pause();
      btnPlay.textContent = '▶';
      estadoPista.textContent = 'Pausado';
    }
  });

  audio.addEventListener('ended', () => {
    btnPlay.textContent = '▶';
    estadoPista.textContent = 'Finalizado';
  });
</script>
